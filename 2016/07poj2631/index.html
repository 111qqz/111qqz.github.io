 42void bfs( int s)
 43{
 44    ms(d,0x3f);
 45    ms(vis,false);
 46    queue<int>q;
 47    q.push(s);
 48    vis[s] = true;
 49    d[s] =  0 ;
 50
 51    while (!q.empty())
 52    {
 53	int u = q.front() ; q.pop();
 54	lst = u ;
 55//	cout<<"u:"<<u<<endl;
 56	int siz = edge[u].size();
 57
 58	for ( int i = 0 ; i < siz ; i++)
 59	{
 60	    int v = edge[u][i].fst;
 61	    if (vis[v]) continue;
 62	    vis[v] = true;
 63	    d[v] = d[u] + edge[u][i].sec;
 64	    ans = max(d[v],ans);
 65	    q.push(v);
 66	}
 67    }
 68
 69    int mx = 0;
 70    for ( int i = 1 ; i <= n ; i++)
 71    {
 72//	cout<<"i:"<<i<<" d[i]:"<<d[i]<<endl;
 73	if (d[i]>mx)
 74	{
 75	    mx = d[i];
 76	    lst = i ;
 77	}
 78    }
 79
 80}
